적용 로직 (의사코드) 
const eligible = coupons.filter(c =>
  (c.allowedPaymentMethods.length===0 || c.allowedPaymentMethods.includes(order.paymentMethod)) &&
  (c.allowedPaymentTypes.length===0   || c.allowedPaymentTypes.includes(order.paymentType))
);
// 중복 불가 → 최대 할인액 자동 선택
const applied = eligible.reduce((best,c) => calcDiscount(c) > calcDiscount(best) ? c : best);
